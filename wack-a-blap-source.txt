const LABELS = ["Q", "W", "E", "A", "S", "D", "Z", "X", "C"] as const;
const HOLE_BY_CODE: Record<string, number> = {
  KeyQ: 0,
  KeyW: 1,
  KeyE: 2,
  KeyA: 3,
  KeyS: 4,
  KeyD: 5,
  KeyZ: 6,
  KeyX: 7,
  KeyC: 8,

const HOLE_BY_KEY: Record<string, number> = {
  q: 0,
  w: 1,
  e: 2,
  a: 3,
  s: 4,
  d: 5,
  z: 6,
  x: 7,
  c: 8,
};

function thump(ms: number) {
  if (typeof navigator !== "undefined" && typeof navigator.vibrate === "function") {
    navigator.vibrate(ms);
  }
}

function holeFromEvent(e: KeyboardEvent) {
  return HOLE_BY_CODE[e.code] ?? HOLE_BY_KEY[e.key.toLowerCase()];
}

const RAMP = 40;
const MAX_FLED = 5;
const BUFFER_MS = 140;
const SLAM = 0.78;
const SAVE_KEY = "wack-blaps-v1";
const MUTE_KEY = "wack-blaps-mute";

const BARKS: Record<Mood, string[]> = {
  idle: ["PRAISE BE.", "MADE IN HIS IMAGE.", "GIFTED WITH FLESH.", "BLESSED WITH PURPOSE."],
  roar: [
    "PRAISE BE TO SPACE KING.",
    "GO, BROTHERS.",
    "CRACKING HEADS.",
    "WE WILL BE REBORN.",
    "ARMORED IN PSYCHOSIS.",
  ],
  annoyed: ["YOU'RE RUNNING OUT OF TIME.", "IT FLED.", "TOO SLOW, BROTHER."],
  laugh: ["ENDLESS WAR.", "PRAISE BE.", "FOR THE SPACE KING."],
};

function pick(mood: Mood) {
  const lines = BARKS[mood];
  return lines[Math.floor(Math.random() * lines.length)] ?? "EYES UP.";
}

function clockLabel(seconds: number) {
  const whole = Math.max(0, Math.floor(seconds));
  const minutes = Math.floor(whole / 60);
  const remain = whole % 60;
  return `${minutes}:${remain.toString().padStart(2, "0")}`;
}

function emptyHole(): Hole {
  return {
    phase: "empty",
    kind: "blap",
    t: 0,
    stay: 1,
    rise: 0.16,
    duck: 0.2,
    splat: 0.42,
    hot: 0,
    slam: 0,
    flash: null,
    flashT: 0,
    strikeAt: 0,
  };
}

function createSim(best: number): Sim {
  return {
    phase: "title",
    reason: "fled",
    elapsed: 0,
    score: 0,
    combo: 0,
    bestCombo: 0,
    crushed: 0,
    fled: 0,
    best,
    prevBest: best,
    newBest: false,
    spawnIn: 0.4,
    mercy: 0,
    kingIn: 8,
    holes: Array.from({ length: 9 }, emptyHole),
    mood: "roar",
    moodT: 0,
    bark: "HOLD THE DECK.",
    idleIn: 3,
    trauma: 0,
    clock: 0,
    freeze: 0,
    floaters: [],
    particles: [],
    reduced: false,
    nextId: 1,
  };
}

function loadBest() {
  try {
    const raw = localStorage.getItem(SAVE_KEY);
    if (!raw) return 0;
    const parsed = JSON.parse(raw) as { best?: number };
    return typeof parsed.best === "number" && parsed.best > 0 ? parsed.best : 0;
  } catch {
    return 0;
  }
}

function tune(elapsed: number) {
  const p = Math.max(0, elapsed) / RAMP;
  const head = Math.min(1, p);
  const ease = 1 - Math.exp(-Math.max(0, p - 1) * 0.55);
  let maxUp = Math.min(4, 1 + Math.floor(head * 2.2 + ease * 1.5));
  if (elapsed < 8) maxUp = 1;
  else maxUp = Math.max(2, maxUp);
  return {
    maxUp,
    stay: Math.max(0.3, 1.2 - head * 0.66 - ease * 0.24),
    rise: Math.max(0.05, 0.16 - head * 0.07 - ease * 0.03),
    duck: Math.max(0.12, 0.2 - ease * 0.05),
    gap: Math.max(0.16, 0.9 - head * 0.56 - ease * 0.18),
  };
}

function hitable(h: Hole) {
  if (h.phase === "rise" || h.phase === "up") return true;
  if (h.phase === "duck") return h.t < h.duck * 0.55;
  return false;
}

function setMood(s: Sim, mood: Mood, time: number) {
  s.mood = mood;
  s.moodT = time;
  s.bark = pick(mood);
  if (mood !== "idle") speak(s.bark);
  if (mood === "idle") s.idleIn = 3.2;
}

function spawnFloat(s: Sim, hole: number, text: string, kind: Floater["kind"]) {
  s.floaters.push({ id: s.nextId++, hole, text, kind, life: 0.7 });
}

function spawnBlood(s: Sim, hole: number) {
  const count = s.reduced ? 4 : 12;
  for (let n = 0; n < count; n++) {
    const ang = -Math.PI * (0.15 + Math.random() * 0.7);
    const sp = 28 + Math.random() * 55;
    s.particles.push({
      id: s.nextId++,
      hole,
      x: (Math.random() - 0.5) * 16,
      y: -8,
      vx: Math.cos(ang) * sp,
      vy: Math.sin(ang) * sp,
      life: 0.32 + Math.random() * 0.28,
      max: 0.55,
      size: 4 + Math.random() * 7,
      kind: Math.random() > 0.75 ? "spark" : "blood",
    });
  }
  if (s.particles.length > 64) s.particles.splice(0, s.particles.length - 64);
}

function crush(s: Sim, index: number) {
  const h = s.holes[index];
  if (!h) return;
  const kind = h.kind;
  h.phase = "splat";
  h.t = 0;
  h.splat = SLAM;
  h.strikeAt = 0;
  h.slam = SLAM;
  h.flash = "hit";
  h.flashT = 0.45;
  s.combo += 1;
  s.bestCombo = Math.max(s.bestCombo, s.combo);
  let gained = 100 + (s.combo - 1) * 25;
  let label = `+${gained}`;
  if (kind === "king") {
    gained *= 3;
    label = `+${gained}`;
  }
  if (kind === "globule") {
    const forgiven = s.fled > 0;
    if (forgiven) s.fled -= 1;
    s.mercy = 3.4;
    gained += 200;
    label = forgiven ? "SAVED" : "SLOW";
  }
  s.score += gained;
  s.crushed += 1;
  s.freeze = s.reduced ? 0 : 0.048;
  s.trauma = Math.min(1, s.trauma + (s.reduced ? 0 : 0.42));
  setMood(s, s.combo >= 8 ? "laugh" : "roar", 0.62);
  spawnFloat(s, index, label, "score");
  spawnBlood(s, index);
  playHit();
  if (!s.reduced) thump(22);
}

function whiff(s: Sim, index: number) {
  const h = s.holes[index];
  if (!h) return;
  h.strikeAt = 0;
  h.slam = SLAM;
  const had = s.combo;
  s.combo = 0;
  h.flash = "whiff";
  h.flashT = 0.16;
  s.trauma = Math.min(1, s.trauma + (s.reduced ? 0 : 0.14));
  setMood(s, "annoyed", 0.45);
  if (had >= 3) spawnFloat(s, index, "COMBO", "bad");
  playWhiff();
  if (!s.reduced) thump(10);
}

function finish(s: Sim) {
  if (s.phase !== "play") return;
  s.phase = "over";
  s.reason = "fled";
  s.newBest = s.score > s.prevBest && s.score > 0;
  if (s.score > s.best) s.best = s.score;
  s.prevBest = s.best;
  try {
    localStorage.setItem(SAVE_KEY, JSON.stringify({ version: 1, best: s.best }));
  } catch {
    /* ignore quota */
  }
  setMood(s, "annoyed", 6);
  stopDrone();
  playOver(false);
}

function escape(s: Sim, h: Hole) {
  s.fled += h.kind === "king" ? 2 : 1;
  s.combo = 0;
  h.flash = "gone";
  h.flashT = 0.22;
  setMood(s, "annoyed", 0.5);
  if (h.kind === "king") s.bark = "THE KING SAW THAT.";
  playEscape();
  if (s.fled >= MAX_FLED) finish(s);
}

function resolveStrikes(s: Sim, now: number) {
  for (let i = 0; i < s.holes.length; i++) {
    const h = s.holes[i];
    if (!h || !h.strikeAt) continue;
    if (hitable(h)) {
      crush(s, i);
      continue;
    }
    if (h.phase === "splat") {
      h.strikeAt = 0;
      continue;
    }
    if (now - h.strikeAt >= BUFFER_MS) whiff(s, i);
  }
}

function step(s: Sim, dt: number, now: number) {
  const diff = tune(s.elapsed);
  resolveStrikes(s, now);
  if (s.phase !== "play") return;

  for (const h of s.holes) {
    if (h.phase === "empty") continue;
    h.t += dt;
    if (h.phase === "rise" && h.t >= h.rise) {
      h.phase = "up";
      h.t = 0;
    } else if (h.phase === "up" && h.t >= h.stay) {
      h.phase = "duck";
      h.t = 0;
      escape(s, h);
      if (s.phase !== "play") return;
    } else if (h.phase === "splat" && h.t >= h.splat) {
      h.phase = "empty";
      h.kind = "blap";
      h.t = 0;
      h.slam = 0;
    } else if (h.phase === "duck" && h.t >= h.duck) {
      h.phase = "empty";
      h.kind = "blap";
      h.t = 0;
    }
  }

  if (s.mercy > 0) s.mercy = Math.max(0, s.mercy - dt);
  if (s.kingIn > 0) s.kingIn -= dt;
  const slow = s.mercy > 0 ? 1.6 : 1;

  s.spawnIn -= dt;
  const alive = s.holes.filter((h) => h.phase !== "empty").length;
  if (s.spawnIn <= 0 && alive < diff.maxUp) {
    const open = s.holes
      .map((h, i) => (h.phase === "empty" ? i : -1))
      .filter((i) => i >= 0);
    if (open.length) {
      const index = open[Math.floor(Math.random() * open.length)] ?? 0;
      const h = s.holes[index];
      if (h) {
        const kingUp = s.holes.some((hole) => hole.kind === "king" && hole.phase !== "empty");
        const globUp = s.holes.some((hole) => hole.kind === "globule" && hole.phase !== "empty");
        h.phase = "rise";
        h.t = 0;
        h.kind = "blap";
        h.rise = Math.max(0.05, diff.rise);
        h.duck = diff.duck;
        h.stay = diff.stay * (0.88 + Math.random() * 0.24) * slow;
        if (!kingUp && s.kingIn <= 0) {
          h.kind = "king";
          h.stay = diff.stay * 0.68;
          s.kingIn = 11;
        } else if (!globUp && Math.random() < 0.09) {
          h.kind = "globule";
          h.stay *= 1.08;
        }
        s.spawnIn = diff.gap * (0.7 + Math.random() * 0.55) * slow;
        playPop();
      }
    } else {
      s.spawnIn = 0.12;
    }
  }

  s.elapsed += dt;
  pressDrone(s.elapsed);
}

function advance(s: Sim, dt: number, now: number) {
  s.clock += dt;
  s.trauma = Math.max(0, s.trauma - dt * 1.85);
  if (s.moodT > 0) {
    s.moodT -= dt;
    if (s.moodT <= 0 && s.phase === "play") {
      s.mood = "idle";
      s.bark = pick("idle");
      s.idleIn = 3.2;
    }
  } else if (s.phase === "play" && s.mood === "idle") {
    s.idleIn -= dt;
    if (s.idleIn <= 0) {
      s.bark = pick("idle");
      s.idleIn = 3.2;
    }
  }

  for (const h of s.holes) {
    h.hot = Math.max(0, h.hot - dt);
    if (h.flashT > 0) {
      h.flashT = Math.max(0, h.flashT - dt);
      if (h.flashT === 0) h.flash = null;
    }
    h.slam = Math.max(0, h.slam - dt);
  }

  for (const f of s.floaters) f.life -= dt;
  s.floaters = s.floaters.filter((f) => f.life > 0);
  for (const p of s.particles) {
    p.life -= dt;
    p.x += p.vx * dt;
    p.y += p.vy * dt;
    p.vy += 110 * dt;
  }
  s.particles = s.particles.filter((p) => p.life > 0);

  if (s.phase !== "play" || document.hidden) return;
  if (s.freeze > 0) {
    s.freeze -= dt;
    return;
  }
  step(s, dt, now);
}

function begin(s: Sim) {
  const best = s.best;
  const reduced = s.reduced;
  const next = createSim(best);
  next.reduced = reduced;
  next.phase = "play";
  next.mood = "roar";
  next.moodT = 0.8;
  next.bark = "FOR THE SPACE KING.";
  next.spawnIn = 0.28;
  Object.assign(s, next);
  s.holes = next.holes;
  s.floaters = [];
  s.particles = [];
  playStart();
  startDrone();
  speak(next.bark);
}

function pose(h: Hole) {
  if (h.phase === "empty") return { pop: 0, frame: 1, sx: 1, sy: 1 };
  if (h.phase === "rise") {
    const p = Math.min(1, h.t / Math.max(0.05, h.rise));
    const e = 1 - (1 - p) ** 3;
    return { pop: e, frame: e < 0.48 ? 1 : 2, sx: 0.94, sy: 1.06 };
  }
  if (h.phase === "up") {
    const bob = Math.sin(h.t * 10);
    return { pop: 1, frame: bob > 0 ? 3 : 2, sx: 1 + bob * 0.035, sy: 1 - bob * 0.03 };
  }
  if (h.phase === "duck") {
    const p = Math.min(1, h.t / Math.max(0.05, h.duck));
    return { pop: 1 - p * p, frame: p < 0.45 ? 2 : 1, sx: 1, sy: 0.96 };
  }
  const p = Math.min(1, h.t / Math.max(0.05, h.splat));
  const pop = p < 0.7 ? 1 : 1 - (p - 0.7) / 0.3;
  const squash = p < 0.12 ? 1.18 : 1;
  return { pop, frame: 4, sx: squash, sy: 1 / squash };
}

function maceFrame(slam: number) {
  const p = 1 - slam / SLAM;
  if (p < 0.1) return 1;
  if (p < 0.2) return 2;
  if (p < 0.82) return 3;
  return 4;
}

const WARRIOR_FRAME: Record<Mood, number> = { idle: 1, roar: 2, annoyed: 3, laugh: 4 };

function queueStrike(s: Sim, index: number, now: number) {
  const h = s.holes[index];
  if (!h || h.strikeAt) return;
  h.strikeAt = now;
  h.hot = 0.14;
}

export function WhackGame() {
  const sim = useRef<Sim>(createSim(0));
  const [tick, setTick] = useState(0);
  const [muted, setMutedState] = useState(false);

  useEffect(() => {
    const best = loadBest();
    sim.current.best = best;
    sim.current.prevBest = best;
    sim.current.reduced = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    const mute = localStorage.getItem(MUTE_KEY) === "1";
    setMutedState(mute);
    setMuted(mute);
    setTick((n) => n + 1);

    const onVis = () => {
      if (document.visibilityState === "visible") resumeAudio();
    };
    document.addEventListener("visibilitychange", onVis);

    let last = performance.now();
    let raf = 0;
    const loop = (now: number) => {
      const dt = Math.min(0.05, (now - last) / 1000);
      last = now;
      advance(sim.current, dt, now);
      setTick((n) => (n + 1) % 100000);
      raf = requestAnimationFrame(loop);
    };
    raf = requestAnimationFrame(loop);

    const onKey = (e: KeyboardEvent) => {
      if (e.repeat) return;
      const s = sim.current;
      const index = holeFromEvent(e);
      const starter = e.code === "Enter" || e.code === "Space" || index !== undefined;
      if (s.phase !== "play") {
        if (!starter) return;
        if (e.code === "Space") e.preventDefault();
        unlockAudio();
        begin(s);
        setTick((n) => n + 1);
        return;
      }
      if (index === undefined) return;
      e.preventDefault();
      unlockAudio();
      queueStrike(s, index, performance.now());
    };
    window.addEventListener("keydown", onKey);
    return () => {
      cancelAnimationFrame(raf);
      window.removeEventListener("keydown", onKey);
      document.removeEventListener("visibilitychange", onVis);
      stopDrone();
    };
  }, []);

  const s = sim.current;
  const mag = s.reduced ? 0 : s.trauma * s.trauma;
  const shake = `translate(${Math.sin(s.clock * 54) * 14 * mag}px, ${Math.cos(s.clock * 46) * 9 * mag}px) rotate(${Math.sin(s.clock * 63) * 1.8 * mag}deg)`;

  const start = () => {
    unlockAudio();
    begin(sim.current);
    setTick((n) => n + 1);
  };

  const toggleMute = () => {
    unlockAudio();
    const next = !isMuted();
    setMuted(next);
    setMutedState(next);
    try {
      localStorage.setItem(MUTE_KEY, next ? "1" : "0");
    } catch {
      /* ignore */
    }
  };

  return (
    <main className="cabinet" data-tick={tick}>
      <p className="sr-only" aria-live="polite">
        {s.phase === "over"
          ? `The blaps scurried. Score ${s.score}. Held ${clockLabel(s.elapsed)}.`
          : s.phase === "play"
            ? `Score ${s.score}. Fled ${s.fled} of ${MAX_FLED}.`
            : "WACK-A-BLAP. Press to start."}
      </p>

      {s.phase === "play" ? (
        <>
          <header className="topbar">
            <div>
              <p className="kicker">Held</p>
              <p className="score-num nums">{clockLabel(s.elapsed)}</p>
            </div>
            <div className="text-center">
              <p className="kicker">Score</p>
              <p className="score-num nums">{s.score.toLocaleString()}</p>
            </div>
            <button type="button" className="ghost-btn" onClick={toggleMute} aria-label={muted ? "Unmute" : "Mute"}>
              {muted ? <VolumeX aria-hidden /> : <Volume2 aria-hidden />}
            </button>
          </header>
          <div className="statrow">
            <div>
              <p className="kicker">Fled {s.fled}/{MAX_FLED}</p>
              <div className="pips" aria-hidden>
                {Array.from({ length: MAX_FLED }, (_, i) => (
                  <span key={i} className={i < s.fled ? "pip on" : "pip"} />
                ))}
              </div>
            </div>
            <p className="bark nums">{s.combo > 1 ? `x${s.combo}  ${s.bark}` : s.bark}</p>
          </div>
          <div className="stage playing">
            <aside className="warrior-card panel">
              <img src={`/sprites/warrior-${WARRIOR_FRAME[s.mood]}.png?v=5`} alt="" draggable={false} />
              <div>
                <p className="kicker">Psycho warrior</p>
                <p className="bark">{s.bark}</p>
                <p className="hint">{s.mercy > 0 ? "Deck slow" : `Crushed ${s.crushed}`}</p>
              </div>
            </aside>
            <div className="board-wrap" style={{ transform: shake }}>
              <div className="board">
                {s.holes.map((h, i) => {
                  const view = pose(h);
                  const bits = s.particles.filter((p) => p.hole === i);
                  const notes = s.floaters.filter((f) => f.hole === i);
                  return (
                    <button
                      key={LABELS[i]}
                      type="button"
                      className="hole"
                      data-up={view.pop > 0.62 ? "1" : "0"}
                      data-flash={h.flash ?? "none"}
                      data-kind={h.phase === "empty" ? "none" : h.kind}
                      style={h.slam > 0 ? { zIndex: 8 } : undefined}
                      aria-label={`Strike hole ${LABELS[i]}`}
                      onPointerDown={(e) => {
                        e.preventDefault();
                        unlockAudio();
                        if (sim.current.phase !== "play") return;
                        queueStrike(sim.current, i, performance.now());
                      }}
                    >
                      <span className="pit" />
                      <span className="blap-clip">
                        {h.phase !== "empty" && (
                          <img
                            className="blap"
                            src={`/sprites/blap-${view.frame}.png?v=3`}
                            alt=""
                            draggable={false}
                            style={{
                              transform: `translateY(${(1 - view.pop) * 78}%) scale(${view.sx}, ${view.sy})`,
                            }}
                          />
                        )}
                      </span>
                      <span className="lip" />
                      {h.slam > 0 && (
                        <img
                          className="mace"
                          src={`/sprites/mace-${maceFrame(h.slam)}.png?v=9`}
                          alt=""
                          draggable={false}
                          style={{
                            transform: `translate(-50%, -58%) scale(${maceFrame(h.slam) === 3 ? 1.18 : 1})`,
                          }}
                        />
                      )}
                      {bits.map((p) => (
                        <span
                          key={p.id}
                          className={`particle ${p.kind}`}
                          style={{
                            left: `calc(50% + ${p.x}px)`,
                            top: `calc(42% + ${p.y}px)`,
                            width: p.size,

let ctx: AudioContext | null = null;
let master: GainNode | null = null;
let sfx: GainNode | null = null;

let noiseBuf: AudioBuffer | null = null;
let muted = false;

function context(): AudioContext | null {
  if (typeof window === "undefined") return null;
  if (!ctx) {
    const AC = window.AudioContext || (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
    if (!AC) return null;
    ctx = new AC({ latencyHint: "interactive" });
    master = ctx.createGain();
    sfx = ctx.createGain();
    sfx.connect(master);
    master.connect(ctx.destination);
    master.gain.value = muted ? 0.0001 : 1;
    const len = Math.floor(ctx.sampleRate * 0.4);
    noiseBuf = ctx.createBuffer(1, len, ctx.sampleRate);
    const data = noiseBuf.getChannelData(0);
    for (let i = 0; i < len; i++) data[i] = Math.random() * 2 - 1;
  }
  return ctx;
}

export function unlockAudio() {
  const audio = context();
  if (audio && audio.state === "suspended") void audio.resume();
}

export function resumeAudio() {
  if (ctx && ctx.state === "suspended") void ctx.resume();
}

export function setMuted(next: boolean) {
  muted = next;
  if (next) hush();
  if (!ctx || !master) return;
  master.gain.setTargetAtTime(next ? 0.0001 : 1, ctx.currentTime, 0.02);
}

export function isMuted() {
  return muted;
}

function burst(freq: number, dur: number, type: OscillatorType, peak: number, slideTo?: number) {
  const audio = context();
  if (!audio || !sfx) return;
  const play = () => {
    if (!sfx) return;
    const osc = audio.createOscillator();
    const gain = audio.createGain();
    osc.type = type;
    const rate = 1 + (Math.random() * 2 - 1) * 0.04;
    osc.frequency.setValueAtTime(freq * rate, audio.currentTime);
    if (slideTo) osc.frequency.exponentialRampToValueAtTime(Math.max(40, slideTo), audio.currentTime + dur);
    gain.gain.setValueAtTime(0.0001, audio.currentTime);
    gain.gain.exponentialRampToValueAtTime(Math.max(0.0002, peak), audio.currentTime + 0.01);
    gain.gain.exponentialRampToValueAtTime(0.0001, audio.currentTime + dur);
    osc.connect(gain);
    gain.connect(sfx);
    osc.start();
    osc.stop(audio.currentTime + dur + 0.02);
    osc.onended = () => {
      osc.disconnect();
      gain.disconnect();
    };
  };
  if (audio.state === "suspended") void audio.resume().then(play);
  else play();
}

function crunch(dur: number, peak: number) {
  const audio = context();
  if (!audio || !sfx || !noiseBuf) return;
  const play = () => {
    if (!sfx || !noiseBuf) return;
    const src = audio.createBufferSource();
    src.buffer = noiseBuf;
    const filter = audio.createBiquadFilter();
    filter.type = "lowpass";
    filter.frequency.setValueAtTime(1400, audio.currentTime);
    filter.frequency.exponentialRampToValueAtTime(140, audio.currentTime + dur);
    const gain = audio.createGain();
    gain.gain.setValueAtTime(0.0001, audio.currentTime);
    gain.gain.exponentialRampToValueAtTime(peak, audio.currentTime + 0.008);
    gain.gain.exponentialRampToValueAtTime(0.0001, audio.currentTime + dur);
    src.connect(filter);
    filter.connect(gain);
    gain.connect(sfx);
    src.start();
    src.stop(audio.currentTime + dur + 0.02);
    src.onended = () => {
      src.disconnect();
      filter.disconnect();
      gain.disconnect();
    };
  };
  if (audio.state === "suspended") void audio.resume().then(play);
  else play();
}

export function startDrone() {
  const audio = context();
  if (!audio || !master || drone) return;
  const play = () => {
    if (!master || drone) return;
    const osc = audio.createOscillator();
    const gain = audio.createGain();
    const filter = audio.createBiquadFilter();
    osc.type = "sawtooth";
    osc.frequency.value = 46;
    filter.type = "lowpass";
    filter.frequency.value = 220;
    gain.gain.value = 0.0001;
    osc.connect(filter);
    filter.connect(gain);
    gain.connect(master);
    osc.start();
    gain.gain.setTargetAtTime(0.028, audio.currentTime, 0.4);
    drone = osc;
    droneGain = gain;
  };
  if (audio.state === "suspended") void audio.resume().then(play);
  else play();
}

export function pressDrone(elapsed: number) {
  if (!ctx || !drone) return;
  const push = Math.min(4, Math.max(0, elapsed) / 40);
  drone.frequency.setTargetAtTime(46 + push * 16, ctx.currentTime, 0.35);

let lastSpoken = 0;

export function speak(line: string) {
  if (muted || typeof window === "undefined" || !window.speechSynthesis) return;
  const now = performance.now();
  if (now - lastSpoken < 900) return;
  lastSpoken = now;
  const utter = new SpeechSynthesisUtterance(line.replace(/\.$/, ""));
  utter.rate = 1.02;
  utter.pitch = 0.68;
  utter.volume = 1;
  const voices = window.speechSynthesis.getVoices();
  const voice =
    voices.find((item) => /en/i.test(item.lang) && /male|daniel|david|alex|fred|guy/i.test(item.name)) ??
    voices.find((item) => /en/i.test(item.lang));
  if (voice) utter.voice = voice;
  window.speechSynthesis.cancel();
  window.speechSynthesis.speak(utter);
}

export function hush() {
  if (typeof window === "undefined" || !window.speechSynthesis) return;
  window.speechSynthesis.cancel();
}

export function stopDrone() {
  if (!ctx || !drone || !droneGain) return;
  droneGain.gain.setTargetAtTime(0.0001, ctx.currentTime, 0.08);
  const osc = drone;
  drone = null;
  window.setTimeout(() => {
    try {
      osc.stop();
      osc.disconnect();
    } catch {
      /* already stopped */
    }
  }, 280);
  droneGain = null;
}
